# SQP01444 — Lookup de Waivers por Tickets

## Contexto

`A2537TKTS` almacena una cadena de tickets concatenados (ej. `1392734138434`).  
`A1672` contiene el detalle de cada ticket con su código de waiver (`A1672CODWA`).  
El join entre ambas tablas no es por clave directa, sino por **búsqueda de subcadena** usando `LOCATE`.

---

## Queries de exploración

### Caso base

```sql
SELECT A2537NCASO, A2537TKTS
FROM PXSAUDIT.A2537
WHERE A2537NCASO = '02124635';
```

### Lookup directo en A1672

```sql
SELECT *
FROM PXSAUDIT.A1672
WHERE A1672CCUST = '139'
  AND A1672CIA   = '139'
  AND A1672FORMA = '2734'
  AND A1672SERIE = '138434';
```

### Flag TIENE_WAIVER por caso (correlated subquery)

```sql
SELECT
    A2537NCASO,
    A2537TKTS,
    (
        SELECT COUNT(*)
        FROM PXSAUDIT.A1672
        WHERE A1672CODWA <> ''
          AND LOCATE(
                TRIM(CHAR(A1672CIA))
                CONCAT TRIM(CHAR(A1672FORMA))
                CONCAT TRIM(CHAR(A1672SERIE)),
                TRIM(A2537TKTS)
              ) > 0
    ) AS TIENE_WAIVER
FROM PXSAUDIT.A2537
WHERE A2537NCASO IN ('02124635','02124636','02124640','02124647');
```

### Casos que tienen al menos un ticket SIN waiver (EXISTS)

```sql
SELECT DISTINCT A.A2537NCASO, A.A2537TKTS
FROM PXSAUDIT.A2537 A
WHERE EXISTS (
    SELECT 1
    FROM PXSAUDIT.A1672 X
    WHERE NULLIF(TRIM(X.A1672CODWA), '') IS NULL
      AND LOCATE(
            TRIM(CHAR(X.A1672CIA)) ||
            TRIM(CHAR(X.A1672FORMA)) ||
            TRIM(CHAR(X.A1672SERIE)),
            TRIM(A.A2537TKTS)
          ) > 0
);
```

### Variante con CTE (A1672_OK)

```sql
WITH A1672_OK AS (
    SELECT
        TRIM(CHAR(A1672CIA)) ||
        TRIM(CHAR(A1672FORMA)) ||
        TRIM(CHAR(A1672SERIE)) AS CODIGO
    FROM PXSAUDIT.A1672
    WHERE A1672CODWA IS NULL
       OR TRIM(A1672CODWA) = ''
)
SELECT DISTINCT A.A2537NCASO, A.A2537TKTS
FROM PXSAUDIT.A2537 A
WHERE EXISTS (
    SELECT 1 FROM A1672_OK X
    WHERE LOCATE(X.CODIGO, TRIM(A.A2537TKTS)) > 0
);
```

> **Nota:** Anteponer/posponer espacio al token y a la cadena (`' ' || CODIGO || ' '`) evita falsos positivos por subcadenas parciales.

### Búsqueda de ticket exacto en A2537TKTS

```sql
SELECT *
FROM PXSAUDIT.A2537
WHERE LOCATE(
    ' ' || '1392143257053' || ' ',
    ' ' || TRIM(A2537TKTS) || ' '
) > 0;
```

### JOIN explícito con ticket encontrado (debug)

```sql
SELECT DISTINCT
    A.A2537NCASO,
    A.A2537TKTS,
    TRIM(CHAR(X.A1672CIA)) ||
    TRIM(CHAR(X.A1672FORMA)) ||
    TRIM(CHAR(X.A1672SERIE)) AS TICKET_ENCONTRADO
FROM PXSAUDIT.A2537 A
JOIN PXSAUDIT.A1672 X
    ON LOCATE(
         TRIM(CHAR(X.A1672CIA)) ||
         TRIM(CHAR(X.A1672FORMA)) ||
         TRIM(CHAR(X.A1672SERIE)),
         ' ' || TRIM(A.A2537TKTS) || ' '
       ) > 0
WHERE NULLIF(TRIM(X.A1672CODWA), '') IS NULL
FETCH FIRST 100 ROWS ONLY;
```

## Stored Procedure — SQP01444

```sql
CREATE OR REPLACE PROCEDURE PXSAUDIT.SQP01444
(
    IN V_CCUST   VARCHAR(3),
    IN V_TICKETS VARCHAR(2000)
)
RESULT SETS 1
LANGUAGE SQL
SPECIFIC PXSAUDIT.SQP01444
BEGIN

    DECLARE V_SQL    VARCHAR(4000);
    DECLARE CURQRY01 CURSOR WITH RETURN FOR STMTQRY01;

    SET V_SQL =
        'SELECT
             TRIM(A1672CCUST) AS A1672CCUST,
             TRIM(A1672CIA) CONCAT TRIM(A1672FORMA) CONCAT TRIM(A1672SERIE) AS TICKET,
             TRIM(A1672CODWA) AS A1672CODWA
         FROM PXSAUDIT.A1672 A
         WHERE A.A1672CCUST = ''' || V_CCUST || '''
           AND A.A1672CODWA <> ''''
           AND LOCATE(
                 '' '' CONCAT TRIM(A1672CIA) CONCAT TRIM(A1672FORMA) CONCAT TRIM(A1672SERIE) CONCAT '' '',
                 '' ' || TRIM(V_TICKETS) || ' '') > 0';

    PREPARE STMTQRY01 FROM V_SQL;
    OPEN CURQRY01;

END
```

IN V_FVETO2 VARCHAR(8),

INOUT IO_PAGROW INTEGER,

INOUT IO_TOTPAG INTEGER,

DECLARE SQLFINAL VARCHAR ( 7000 ) ;

DECLARE CURQRY01 CURSOR FOR STMTQRY01 ;

DECLARE C_CNT CURSOR FOR STMTCNT ;

  

/* =========================

BASE SQL

========================= */

SET V_SQL =

'SELECT

ROW_NUMBER() OVER(ORDER BY A.A2537SEQ) AS RN,

COUNT(1) OVER() AS TOTAL_ROWS,

TRIM(A.A2537CCUST) AS A2537CCUST,

TRIM(A.A2537SEQ) AS A2537SEQ,

TRIM(A.A2537NCASO) AS A2537NCASO,

TRIM(A.A2537PCASO) AS A2537PCASO,

A.A2537FCRRE,

A.A2537HCRRE,

TRIM(A.A2537NVLO) AS A2537NVLO,

A.A2537FVLO,

TRIM(A.A2537SCPTO) AS A2537SCPTO,

A.A2537FMODI,

A.A2537HMODI,

CW.TKTS_CODWA

LEFT JOIN (

SELECT

B.A1672CCUST,

LISTAGG(TRIM(B.A1672CIA) CONCAT TRIM(B.A1672FORMA) CONCAT TRIM(B.A1672SERIE), '' '')

WITHIN GROUP (ORDER BY B.A1672SERIE) AS TKTS_CODWA,

TRIM(B.A1672CIA) CONCAT TRIM(B.A1672FORMA) CONCAT TRIM(B.A1672SERIE) AS TKT_KEY

FROM PXSAUDIT.A1672 B

WHERE B.A1672CCUST = ''' CONCAT V_CCUST CONCAT '''

AND B.A1672CODWA <> ''''

GROUP BY B.A1672CCUST,

TRIM(B.A1672CIA) CONCAT TRIM(B.A1672FORMA) CONCAT TRIM(B.A1672SERIE)

) CW ON CW.A1672CCUST = A.A2537CCUST

AND A.A2537TKTS LIKE ''%'' CONCAT CW.TKT_KEY CONCAT ''%''

WHERE A.A2537CCUST = ''' CONCAT V_CCUST CONCAT '''' ;

FILTROS

========================= */

IF V_NCASO <> '' THEN

SET V_SQL = V_SQL CONCAT

' AND A2537NCASO = ''' CONCAT V_NCASO CONCAT '''' ;

IF V_TKTS <> '' THEN

SET V_SQL = V_SQL CONCAT

' AND A2537TKTS LIKE ''%' CONCAT TRIM ( V_TKTS ) CONCAT '%''' ;

SET V_SQL = V_SQL CONCAT

' AND A2537CODIT = ''' CONCAT V_CODIT CONCAT '''' ;

' AND A2537FCRRE BETWEEN ''' CONCAT V_FCRRE1 CONCAT ''' AND ''' CONCAT V_FCRRE2 CONCAT '''' ;

CONTAR TOTAL DE FILAS

PAGINACIÓN

========================= */

SET IO_PAGNUM = PRAXIS . F0001 ( 'PAGNUM' , IO_PAGNUM , IO_PAGROW , IO_TOTPAG , IO_TOTROW ) ;

' CONCAT V_SQL CONCAT

') RS

' FETCH FIRST ' CONCAT CHAR ( IO_PAGROW ) CONCAT ' ROWS ONLY' ;
