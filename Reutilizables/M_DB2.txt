-- Buscar un Stored Procedure
SELECT ROUTINE_NAME, SPECIFIC_NAME, EXTERNAL_NAME
FROM QSYS2.SYSPROCS
WHERE ROUTINE_SCHEMA = 'PRAXIS'
  AND SPECIFIC_NAME = 'PX172S02A1838';

-- Buscar parametros de un Stored Procedure (version 1: todos los campos)
SELECT SPECIFIC_SCHEMA, SPECIFIC_NAME, ORDINAL_POSITION, PARAMETER_MODE,
       PARAMETER_NAME, DATA_TYPE, CHARACTER_MAXIMUM_LENGTH
FROM QSYS2.SYSPARMS
WHERE SPECIFIC_SCHEMA = 'PRAXIS'
  AND SPECIFIC_NAME = 'SQP05424'
ORDER BY ORDINAL_POSITION;

-- Buscar parametros de un Stored Procedure (version 2: resumida)
SELECT SPECIFIC_SCHEMA AS ESQUEMA, SPECIFIC_NAME AS NOMBRE_SP,
       PARAMETER_NAME AS PARAMETRO, DATA_TYPE AS TIPO
FROM QSYS2.SYSPARMS
WHERE SPECIFIC_NAME = 'PX095S01A1691';

-- Variables para prueba de un SP
CREATE OR REPLACE VARIABLE PRAXISMP.TES_STATUS INTEGER;
CREATE OR REPLACE VARIABLE PRAXISMP.TES_MSG VARCHAR(300);
VALUES PRAXISMP.TES_MSG;   -- para ver la respuesta

-- Eliminar variables de prueba
DROP VARIABLE PRAXISMP.TES_STATUS;
DROP VARIABLE PRAXISMP.TES_MSG;

-- Ejemplo de llamada a un Stored Procedure
CALL PRAXISMP.MPS245('C', '', '134', '', '1', 'pRUEBA', '', PRAXISMP.TES_STATUS, PRAXISMP.TES_MSG);
